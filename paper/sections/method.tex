\section{Method}
\label{sec:method}

We describe Sparse Feature Preservation in three parts: the empirical hypothesis that motivates the method (\cref{sec:hypothesis}), the preservation algorithm (\cref{sec:algorithm}), and the formally verified mathematical properties (\cref{sec:formal}).

\subsection{Hypothesis: Forgetting is Low-Dimensional}
\label{sec:hypothesis}

Let $f_\theta$ be a language model with parameters $\theta$, and let $a_\ell(\theta) \in \R^d$ denote the mean-pooled activation vector at layer $\ell$ for a given input.
When we fine-tune from $\theta^*$ (anchor) to $\theta'$ (fine-tuned), the activation drift at layer $\ell$ is $\Delta a_\ell = a_\ell(\theta') - a_\ell(\theta^*)$.

Let $U \in \R^{d \times k}$ be a PCA basis computed from old-task activations at layer $\ell$, and let $U_r \in \R^{d \times r}$ ($r \ll k \ll d$) be the submatrix of the top-$r$ most important dimensions ranked by ablation.
The projected drift in the important subspace is $\PW(\Delta a_\ell) = U_r U_r^\top \Delta a_\ell$, where $W = \text{col}(U_r)$.

\begin{hypothesis}[H1: Low-Dimensional Forgetting]
\label{hyp:h1}
There exists a constant $C > 0$ such that old-task performance degradation satisfies
\begin{equation}
    \text{forgetting} \leq C \cdot \| \PW(\Delta a) \|,
    \label{eq:h1}
\end{equation}
and the projected drift $\| \PW(\Delta a) \|$ using the top-$r$ importance-ranked dimensions is a better predictor of forgetting than total drift $\| \Delta a \|$, random-dimension drift, or bottom-ranked drift.
\end{hypothesis}

This hypothesis is \emph{falsifiable}: we pre-register four predictors (top-$r$, total, random, bottom-$r$) and require the ranking top $>$ total $>$ random $>$ bottom in $R^2$ for predicting forgetting across training checkpoints.

\paragraph{Ablation importance.}
Given the PCA basis $U = [u_1, \ldots, u_k]$, the importance of dimension $i$ is the increase in old-task loss when that dimension is zeroed out:
\begin{equation}
    \text{imp}(i) = \mathcal{L}_{\text{old}}\!\left(a_\ell - (u_i^\top a_\ell)\, u_i\right) - \mathcal{L}_{\text{old}}(a_\ell),
    \label{eq:importance}
\end{equation}
averaged over the memory set.
The top-$r$ dimensions are those with the largest importance scores.

\subsection{SFP Algorithm}
\label{sec:algorithm}

SFP operates in two phases: \emph{basis construction} (before training) and \emph{preservation} (during training).

\paragraph{Phase 1: Basis construction.}
Given a memory set $\mathcal{M}$ of old-task examples (typically 128), we:
(1)~collect activations at probe layers $\mathcal{L} = \{L/4, L/2, 3L/4\}$;
(2)~compute PCA basis $U_\ell \in \R^{d \times k}$ per layer;
(3)~compute ablation importance per dimension;
(4)~select top-$r$ dimensions, yielding $U_{r,\ell}$;
(5)~record reference activations $a_\ell^* = a_\ell(\theta^*)$.

\paragraph{Phase 2: Preservation during training.}
The training loss combines the new-task objective with a preservation penalty:
\begin{equation}
    \mathcal{L}(\theta) = \Lnew(\theta) + \lambda \sum_{\ell \in \mathcal{L}} \underbrace{\| U_{r,\ell}^\top a_\ell(\theta) - U_{r,\ell}^\top a_\ell(\theta^*) \|^2}_{\mathcal{L}_{\text{pres},\ell}(\theta)},
    \label{eq:sfp_loss}
\end{equation}
where $\lambda > 0$ controls the stability--plasticity trade-off.
By construction, the model is free to change activations in the complement $W_\ell^\perp$.

\paragraph{Gradient-informed variant.}
As an alternative to PCA, we compute $G = \partial \mathcal{L}_{\text{old}} / \partial a_\ell$ on the memory set and take the top-$r$ right singular vectors of $G$ as the preservation basis.
This directly identifies directions the old-task loss is most sensitive to.

\begin{algorithm}[t]
\caption{Sparse Feature Preservation}
\label{alg:sfp}
\begin{algorithmic}[1]
\REQUIRE Model $f_{\theta^*}$, memory set $\mathcal{M}$, new data $\mathcal{D}$, layers $\mathcal{L}$, dims $k$, $r$, weight $\lambda$
\STATE \textbf{// Phase 1: Basis construction}
\FOR{$\ell \in \mathcal{L}$}
    \STATE Collect activations $A_\ell = \{a_\ell(\theta^*, x) : x \in \mathcal{M}\}$
    \STATE $U_\ell \leftarrow \text{PCA}(A_\ell, k)$; rank dims by ablation importance
    \STATE $U_{r,\ell} \leftarrow$ top-$r$ columns; store $a_\ell^*$
\ENDFOR
\STATE \textbf{// Phase 2: Fine-tuning}
\FOR{each batch $B \sim \mathcal{D}$}
    \STATE $\mathcal{L} \leftarrow \Lnew(B; \theta) + \lambda \sum_\ell \| U_{r,\ell}^\top a_\ell(\theta) - U_{r,\ell}^\top a_\ell^* \|^2$
    \STATE Update $\theta$ via gradient descent
\ENDFOR
\end{algorithmic}
\end{algorithm}

\subsection{Formally Verified Properties}
\label{sec:formal}

We formalize SFP's mathematical properties in Lean~4 using Mathlib (v4.17.0).
The formalization comprises approximately 1{,}200 lines of Lean with \textbf{zero} \texttt{sorry} axioms.

\begin{definition}[Preservation Loss, \texttt{preservationLoss}]
\label{def:pres_loss}
For an orthogonal projection $\PW$ onto subspace $W$,
$\Lpres = \| \PW(\Delta a) \|^2$.
\end{definition}

\begin{proposition}[\texttt{preservationLoss\_nonneg}, \texttt{preservationLoss\_eq\_zero\_iff}]
\label{prop:loss_props}
$\Lpres \geq 0$, with equality iff $\Delta a \in W^\perp$.
\end{proposition}

\begin{proposition}[Pythagorean Decomposition, \texttt{drift\_pythagorean\_decomposition}]
\label{prop:pythag}
Total drift decomposes as
$\| \Delta a \|^2 = \Lpres + \| P_{W^\perp}(\Delta a) \|^2$.
Hence $\Lpres \leq \| \Delta a \|^2$: preserving the important subspace is strictly easier than preventing all drift.
\end{proposition}

\begin{proposition}[Gradient Orthogonality, \texttt{gradient\_orthogonality}]
\label{prop:grad_orth}
At a stationary point of $\Lnew + \lambda \Lpres$ where $\Lpres = 0$, the new-task gradient satisfies $\PW(\nabla \Lnew) = 0$, \ie $\nabla \Lnew \in W^\perp$.
New-task learning occurs only in directions orthogonal to the preserved features.
\end{proposition}

\begin{theorem}[Main Reduction, \texttt{sfp\_reduces\_forgetting}]
\label{thm:main}
Assume H1 (\cref{hyp:h1}): $\text{forgetting} \leq C \cdot \| \PW(\Delta a) \|$ for $C \geq 0$.
If $\Lpres \leq \delta$, then
\begin{equation}
    \text{forgetting} \leq C \sqrt{\delta}.
    \label{eq:reduction}
\end{equation}
\end{theorem}

\begin{proof}[Proof sketch]
$\Lpres = \| \PW(\Delta a) \|^2 \leq \delta$ implies $\| \PW(\Delta a) \| \leq \sqrt{\delta}$.
By H1, $\text{forgetting} \leq C\sqrt{\delta}$.
The full Lean proof verifies that $\PW$ is an orthogonal projection (idempotent, self-adjoint, norm-contractive).
\end{proof}

\begin{theorem}[Stability--Plasticity Trade-off, \texttt{stability\_plasticity\_tradeoff}]
\label{thm:tradeoff}
Under the same conditions, (i)~forgetting $\leq C\sqrt{\delta}$ (stability), and (ii)~$\| P_{W^\perp}(\Delta a) \|^2 \geq \| \Delta a \|^2 - \delta$ (plasticity).
As $\lambda \to \infty$, forgetting vanishes but learning is confined to $W^\perp$.
\end{theorem}

\paragraph{Scope of verification.}
The formalization verifies the mathematical chain from H1 to the forgetting bound.
It does \emph{not} verify H1 itself---that is an empirical claim tested in \cref{sec:experiments}.
This separation is deliberate: the verification makes the logical structure transparent, so the reader knows exactly which claim requires empirical evidence.
